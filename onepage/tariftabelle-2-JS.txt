/* ═══ Zentrale Preisliste ═══════════════════════════════════════
   Wortgleich mit den drei Rechner-Feldern. Wer zuerst läuft, baut
   sie — die anderen benutzen dieselbe Instanz und es wird nur ein
   einziges Mal geladen. Darum darf hier nichts abweichen. */
window.AVJ_PREISE = window.AVJ_PREISE || (function(){
  "use strict";
  var URL_JSON  = "https://avj-tools.rent-in-nom.de/preise.json";
  var CACHE_KEY = "avjWebPreise_v1";
  var daten = null, fertig = false, warteschlange = [];

  function pruefe(d){
    if(!d || typeof d !== "object") return false;

    /* Frueher stand hier "neunsitzer und transporter muessen da sein".
       Das war zu eng: sobald ein Rechner eine andere Gruppe liest, haengt
       die Gueltigkeit der ganzen Datei an zwei fremden Abschnitten.
       Jetzt reicht: mindestens eine Gruppe mit mindestens einem Fahrzeug. */
    var gefunden = 0;
    for(var g0 in d){
      if(g0 === "version" || g0 === "stand" || g0 === "erzeugtVon") continue;
      if(d[g0] && typeof d[g0] === "object"){
        for(var k0 in d[g0]){ gefunden++; break; }
      }
    }
    if(!gefunden) return false;
    var ok = true;

    function zahlenreihe(a, laenge){
      if(!a || a.length !== laenge) return false;
      for(var i=0;i<a.length;i++){
        if(typeof a[i] !== "number" || !isFinite(a[i]) || a[i] < 0) return false;
      }
      return true;
    }
    function paarliste(a){
      if(!a || !a.length) return false;
      for(var i=0;i<a.length;i++){
        if(!a[i] || a[i].length !== 2) return false;
        if(typeof a[i][0] !== "number" || typeof a[i][1] !== "number") return false;
        if(!isFinite(a[i][0]) || !isFinite(a[i][1]) || a[i][0] < 0 || a[i][1] < 0) return false;
      }
      return true;
    }

    for(var g in d){
      if(g === "version" || g === "stand" || g === "erzeugtVon") continue;
      if(!d[g] || typeof d[g] !== "object") continue;
      for(var k in d[g]){
        var c = d[g][k];
        if(!c || typeof c !== "object"){ ok = false; continue; }
        if(!zahlenreihe(c.tier, 7))   ok = false;
        if(!zahlenreihe(c.tierKm, 7)) ok = false;
        if(c.dayMoDo && !paarliste(c.dayMoDo)) ok = false;
        if(c.dayFrSa && !paarliste(c.dayFrSa)) ok = false;
        if(c.weekend && !paarliste(c.weekend)) ok = false;
        if(c.sb){
          if(!c.sb.length) ok = false;
          for(var s=0;s<(c.sb||[]).length;s++){
            if(!c.sb[s] || !zahlenreihe(c.sb[s].tier, 7)) ok = false;
          }
        }
      }
    }
    return ok;
  }

  function melde(){
    fertig = true;
    var f = warteschlange; warteschlange = [];
    for(var i=0;i<f.length;i++){ try{ f[i](daten); }catch(e){} }
  }

  function ausCache(){
    try{
      var raw = window.localStorage.getItem(CACHE_KEY);
      if(!raw) return false;
      var d = JSON.parse(raw);
      if(!pruefe(d)) return false;
      daten = d;
      return true;
    }catch(e){ return false; }
  }

  function start(){
    /* Zeitstempel je Minute: hebelt den Zwischenspeicher von GitHub Pages
       aus, ohne ihn ganz abzuschalten. */
    var url = URL_JSON + "?t=" + Math.floor(new Date().getTime()/60000);
    var erledigt = false;
    var frist = setTimeout(function(){
      if(erledigt) return;
      erledigt = true; ausCache(); melde();
    }, 3000);

    try{
      var x = new XMLHttpRequest();
      x.open("GET", url, true);
      x.onreadystatechange = function(){
        if(x.readyState !== 4 || erledigt) return;
        erledigt = true; clearTimeout(frist);
        var d = null;
        try{ d = JSON.parse(x.responseText); }catch(e){}
        if(x.status >= 200 && x.status < 400 && pruefe(d)){
          daten = d;
          try{ window.localStorage.setItem(CACHE_KEY, JSON.stringify(d)); }catch(e){}
        } else {
          ausCache();
        }
        melde();
      };
      x.onerror = function(){
        if(erledigt) return;
        erledigt = true; clearTimeout(frist); ausCache(); melde();
      };
      x.send();
    }catch(e){
      clearTimeout(frist);
      if(!erledigt){ erledigt = true; ausCache(); melde(); }
    }
  }
  start();

  return {
    bereit: function(fn){
      if(fertig){ try{ fn(daten); }catch(e){} }
      else warteschlange.push(fn);
    },
    daten:   function(){ return daten; },
    /* Kaution: kommt aus preise.json, falls dort hinterlegt — sonst 300 */
    kaution: function(){
      var k = daten && daten.kaution;
      return (typeof k === "number" && isFinite(k) && k > 0) ? k : 300;
    }
  };
})();

    neunsitzer:  { oeffnen:"avjOpen",  wahl:".avj-cars"  },

  /* ─── Formatierung ───────────────────────────────────────────── */
  function eur(n){
    return Number(n).toLocaleString("de-DE",
      { minimumFractionDigits:2, maximumFractionDigits:2 }) + " €";
  }
  function kmTxt(n){ return Number(n).toLocaleString("de-DE") + " km inkl."; }
  function esc(s){
    return String(s).replace(/&/g,"&amp;").replace(/</g,"&lt;").replace(/>/g,"&gt;");
  }
  /* Im Titel markiert *Sternchen* den Modellnamen — der wird rot,
     genau wie auf den bisherigen Tabellen. */
  function titelHtml(t){
    var teile = esc(t).split("*"), h = "";
    for(var i=0;i<teile.length;i++){
      h += (i % 2) ? '<span class="rot">' + teile[i] + '</span>' : teile[i];
    }
    return h;
  }
  function ahkTxt(v){
    return v ? ' <span class="kl">(AHK ' + eur(v) + ')</span>' : '';
  }
  function zelle(kl, inhalt, stil){
    return '<div class="avjtab-z ' + kl + '"' + (stil ? ' style="' + stil + '"' : '')
         + '>' + inhalt + '</div>';
  }
  function wert(kmv, preis){
    return '<div class="avjtab-km">' + kmTxt(kmv) + '</div>'
         + '<div class="avjtab-pr">' + eur(preis) + '</div>';
  }
  /* ─── Langzeit ───────────────────────────────────────────────────
     Wortgleiche Regel wie im Tool und in den Rechnern: vier Wochen
     sind der zweite Anker, darueber gibt es keinen Automatikpreis.
     Fehlt der Anker beim Fahrzeug, wird er geschaetzt — sonst wuerde
     die Tabelle etwas anderes zeigen als der Rechner rechnet. */
  var LZ_TAGE = 28;
  function mRund5(n){  return Math.max(5,  Math.round(n / 5)  * 5);  }
  function mRund50(n){ return Math.max(50, Math.round(n / 50) * 50); }
  function lzPreis(c){
    if(c.monthPrice > 0) return c.monthPrice;
    return mRund5((c.tier ? c.tier[6] : 0) * 3);
  }
  function lzKm(c){
    if(c.monthKm > 0) return c.monthKm;
    return mRund50((c.tierKm ? c.tierKm[6] : 0) * 2);
  }
  function lzSb(stufe){
    if(!stufe || !stufe.tier) return 0;
    if(stufe.monat !== undefined && stufe.monat !== null) return stufe.monat;
    return Math.round(stufe.tier[6] * 3);
  }

  function extraKm(c){
    if(AVJTAB_GEPLANT && c.planKm && c.overKm && c.planKm !== c.overKm){
      return '<div style="font-size:12.5px;line-height:1.45">vorab geplant<br><b>'
        + eur(c.planKm) + '</b><br>bei Rückgabe<br><b>' + eur(c.overKm) + '</b></div>';
    }
    return eur(c.overKm || c.planKm || 0);
  }

  /* ─── Daten holen ────────────────────────────────────────────── */

  /* Angaben, die nur die Tabelle braucht und der Rechner nicht. Sie
     sind erst mit Build 34/35 dazugekommen — eine preise.json, die
     vorher erzeugt wurde, kennt sie noch nicht. Fehlen sie dort, wird
     der Wert aus dem Code genommen, statt die Kurzzeittarife und die
     AHK-Zeilen stillschweigend verschwinden zu lassen.

     Preise werden so NIE ersetzt: tier, tierKm, dayMoDo, dayFrSa,
     weekend und sb kommen immer vom Server, wenn er antwortet. */
  var NACHTRAG = ["kurzzeit", "ahk", "kaution"];

  function hole(pfad){
    var t = String(pfad || "").split("."), gruppe = t[0], key = t[1];
    var d = null;
    try{ d = AVJ_PREISE.daten(); }catch(e){}
    var eingebaut = (EINGEBAUT[gruppe] && EINGEBAUT[gruppe][key]) || null;
    var quelle = (d && d[gruppe] && d[gruppe][key]) ? d[gruppe][key] : null;
    var stand  = (d && d.stand) ? d.stand : STAND_FALLBACK;

    if(quelle && eingebaut){
      var kopie = {}, f;
      for(f in quelle) kopie[f] = quelle[f];
      for(var i=0;i<NACHTRAG.length;i++){
        f = NACHTRAG[i];
        if(kopie[f] === undefined && eingebaut[f] !== undefined) kopie[f] = eingebaut[f];
      }
      quelle = kopie;
    }
    if(!quelle && eingebaut){
      quelle = eingebaut;
      stand  = STAND_FALLBACK;
    }
    return quelle ? { c:quelle, stand:stand, gruppe:gruppe, key:key } : null;
  }

  function tagestarife(c){

    var reihen  = hatFrSa ? 3 : 2;

    var h = '<div class="avjtab-reihe">'
      + '<div class="avjtab-z avjtab-band">Tagestarife' + ahkTxt(ahk.tag) + '</div>'
      + zelle('avjtab-rot','je extra km') + '</div>';

    h += '<div class="avjtab-reihe" style="grid-template-rows:'
      + (reihen === 3 ? 'auto auto auto' : 'auto auto') + '">';
    h += zelle('avjtab-gelb', hatFrSa ? '1 Tag<br><br>Mo. - Do.' : '1 Tag',
               'grid-column:1;grid-row:1');
    for(var i=0;i<c.dayMoDo.length;i++){
      h += zelle('avjtab-grau', wert(c.dayMoDo[i][0], c.dayMoDo[i][1]),
                 'grid-column:' + (i+2) + ';grid-row:1');
    }
    h += zelle('avjtab-grau', extraKm(c), 'grid-column:6;grid-row:1/' + (reihen+1));
    var z = 2;
    if(hatFrSa){
      h += zelle('avjtab-gelb','1 Tag<br><br>Fr. - Sa.','grid-column:1;grid-row:2');
      for(var j=0;j<c.dayFrSa.length;j++){
        h += zelle('avjtab-grau', wert(c.dayFrSa[j][0], c.dayFrSa[j][1]),
                   'grid-column:' + (j+2) + ';grid-row:2');
      }
      z = 3;
    }
    h += zelle('avjtab-gelb','7 Tage'
      + (ahk.woche ? '<br><span class="kl2">(AHK ' + eur(ahk.woche) + ')</span>' : ''),
      'grid-column:1;grid-row:' + z);
    h += zelle('avjtab-grau', wert(c.tierKm[6], c.tier[6]), 'grid-column:2/6;grid-row:' + z);
    h += '</div>';
    return h;
  }

  /* Fahrzeuge ohne Tagespaketraster — Low Budget — bekommen die
     Staffel 1 bis 7 Tage und, falls hinterlegt, den Monatspreis. */
  function staffel(c){
    if(c.dayMoDo && c.dayMoDo.length) return "";
    var h = '<div class="avjtab-reihe">'
      + '<div class="avjtab-z avjtab-band">Mietdauer</div>'
      + zelle('avjtab-rot','je extra km') + '</div>';
    h += '<div class="avjtab-staffel">';
    h += '<div class="avjtab-z avjtab-gelb"></div>';
    for(var i=0;i<7;i++){
      h += '<div class="avjtab-z avjtab-gelb">' + (i+1) + (i ? ' Tage' : ' Tag') + '</div>';
    }
    h += '<div class="avjtab-z avjtab-gelb links"><b>Preis</b></div>';
    for(var p=0;p<7;p++){
      h += '<div class="avjtab-z avjtab-grau"><b>' + eur(c.tier[p]) + '</b></div>';
    }
    h += '<div class="avjtab-z avjtab-gelb links"><b>inklusive</b></div>';
    for(var k=0;k<7;k++){
      h += '<div class="avjtab-z avjtab-grau">' + kmTxt(c.tierKm[k]) + '</div>';
    }
    h += '</div>';
    h += '<div class="avjtab-reihe" style="grid-template-columns:1.55fr 1fr">'
      + zelle('avjtab-gelb links','<b>je extra km</b>')
      + zelle('avjtab-grau', extraKm(c)) + '</div>';
    return h;
  }

  function wochenende(c){
    if(!c.weekend || !c.weekend.length) return "";
    var ahk = c.ahk || {};
    var h = '<div class="avjtab-reihe">'
      + '<div class="avjtab-z avjtab-band">Wochenendtarife' + ahkTxt(ahk.we) + '</div>'
      + zelle('avjtab-rot','je extra km') + '</div>';
    h += '<div class="avjtab-reihe">';
    h += zelle('avjtab-gelb','ab<br>Freitag 12:00 Uhr<br>bis<br>Montag 10:00 Uhr','grid-column:1');
    /* Vier Wertspalten stehen zur Verfuegung: 2,3,4,5. Hat das Fahrzeug
       weniger Stufen, wird die erste breiter — die extra-km-Spalte
       bleibt unangetastet. */
    var we = c.weekend, frei = 4 - we.length, sp = 2;
    if(frei < 0) frei = 0;
    for(var w=0; w<we.length && sp < 6; w++){
      var breite = (w === 0 ? frei + 1 : 1);
      if(sp + breite > 6) breite = 6 - sp;
      h += zelle('avjtab-grau', wert(we[w][0], we[w][1]),
                 'grid-column:' + sp + '/' + (sp + breite));
      sp += breite;
    }
    h += zelle('avjtab-grau', extraKm(c), 'grid-column:6');
    h += '</div>';
    return h;
  }

  function haftung(c){
    if(!c.sb || !c.sb.length) return "";
    var n = c.sb.length;
    var raster = 'grid-template-columns:1.55fr repeat(' + n + ',1fr)';
    var h = '<div class="avjtab-luecke"></div>'
      + '<div class="avjtab-reihe"><div class="avjtab-z avjtab-blau">'
      + 'Haftungsreduzierung pro Anmietung</div></div>';

    h += '<div class="avjtab-reihe" style="' + raster + '">'
      + zelle('avjtab-gelb links','<span><b>Selbstbeteiligung</b> '
        + '<span class="kl">je Schadenfall</span></span>');
    for(var s=0;s<n;s++){
      var l = String(c.sb[s].label || "");
      if(l.indexOf("SB") !== 0) l = "SB " + l;
      /* Die drei Sätze sind Aufpreise auf den Grundtarif. Der erste ist
         immer 0 — er ist im Preis enthalten. Bei Kurzzeittarifen gibt es
         keine Reduzierung, das steht wie bisher an der Spalte. */
      var frei = !c.sb[s].we && !(c.sb[s].tier && c.sb[s].tier[0]);
      h += zelle('avjtab-rot', l
        + (!frei && c.kurzzeit ? ' <span class="kl2">nicht bei KZ-Tarif</span>' : ''));
    }
    h += '</div>';

    var zeilen = [['pro Tag', 0]];
    if(c.weekend && c.weekend.length) zeilen.push(['pro Wochenende (WE-Tarif)', 'we']);
    zeilen.push(['pro Woche (7-Tagestarif)', 6]);
    zeilen.push(['pro Monat (4 Wochen)', 'monat']);

    for(var z=0;z<zeilen.length;z++){
      h += '<div class="avjtab-reihe" style="' + raster + '">'
        + zelle('avjtab-gelb links','<b>' + zeilen[z][0] + '</b>');
      for(var q=0;q<n;q++){
        var feld = zeilen[z][1], v;
        if(feld === 'we')         v = c.sb[q].we;
        else if(feld === 'monat') v = lzSb(c.sb[q]);
        else                      v = (c.sb[q].tier || [])[feld];
        h += zelle('avjtab-grau', v ? eur(v) : 'inklusive');
      }
      h += '</div>';
    }
    return h;
  }

  /* ─── Zeichnen ───────────────────────────────────────────────── */
  function zeichne(el){
    var pfad = el.getAttribute("data-fz");
    var d = hole(pfad);
    if(!d){ el.innerHTML = stoerung(); el.setAttribute("data-fertig","stoerung"); return; }
    var c = d.c;
    var titel = el.getAttribute("data-titel");
    if(!titel){
      titel = (c.name || d.key) + (c.example ? " — " + c.example : "");
    }

    var h = '<div class="avjtab-blatt"><div class="avjtab-in">';
    h += kopf(titelHtml(titel));
    h += hinweisRechner(d.gruppe, d.key, c.name || d.key);
    h += kurzzeit(c);
    h += staffel(c);
    h += tagestarife(c);
    h += wochenende(c);
    h += langzeit(c);
    h += haftung(c);
    h += kaution(c);
    h += fuss(d.stand);
    h += '</div></div>';
    el.innerHTML = h;
    el.setAttribute("data-fertig","1");
  }

  function alle(){
    var l = document.querySelectorAll(".avjtab");
    for(var i=0;i<l.length;i++){
      try{ zeichne(l[i]); }
      catch(e){ try{ l[i].innerHTML = stoerung(); }catch(e2){} }
    }
  }

  /* Der Rechner-Knopf: Popup schließen, Rechner öffnen, Fahrzeug wählen */
  document.addEventListener("click", function(e){
    var a = e.target.closest ? e.target.closest("[data-rechner]") : null;
    if(!a) return;
    e.preventDefault();
    var r = RECHNER[a.getAttribute("data-rechner")];
    if(!r) return;
    var knopf = document.getElementById(r.oeffnen);
    if(!knopf) return;
    /* das Popup, in dem die Tabelle steht, schließen — sonst liegen
       zwei Overlays übereinander */
    try{
      var offen = a.closest("[data-avjtab-popup]");
      if(offen) offen.click();
    }catch(e3){}
    knopf.click();
    var key = a.getAttribute("data-fzkey");
    setTimeout(function(){
      try{
        var box = document.querySelector(r.wahl);
        if(!box) return;
        var b = box.querySelector('[data-car="' + key + '"]');
        if(b) b.click();
      }catch(e4){}
    }, 60);
  });

  /* Einmal sofort, einmal wenn die Preise da sind, und danach für
     alles, was Onepage nachträglich in die Seite hängt. */
  function start(){
    alle();
    try{ AVJ_PREISE.bereit(function(){ alle(); }); }catch(e){}
    try{
      if(window.MutationObserver){
        var mo = new MutationObserver(function(){
          var l = document.querySelectorAll(".avjtab:not([data-fertig])");
          if(l.length) alle();
        });
        mo.observe(document.body, { childList:true, subtree:true });
      }
    }catch(e){}
  }
  if(document.readyState === "loading"){
    document.addEventListener("DOMContentLoaded", start);
  } else { start(); }

  return { zeichne: zeichne, alle: alle };
})();
