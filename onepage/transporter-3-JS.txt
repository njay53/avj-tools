/* ═══ Zentrale Preisliste ═══════════════════════════════════════
   Holt preise.json von avj-tools.rent-in-nom.de. Drei Stufen:
     1. Server
     2. der zuletzt gemerkte Stand aus diesem Browser
     3. die Werte, die weiter unten fest im Code stehen
   Damit kann der Rechner nie kaputtgehen, nur veralten.

   Beide Rechner auf dieser Seite teilen sich diese eine Instanz — das
   "window.AVJ_PREISE ||" davor sorgt dafuer, dass nur einmal geladen wird,
   egal welches der beiden Felder zuerst ausgefuehrt wird.

   Bewusst XMLHttpRequest statt fetch: laeuft auch in aelteren Safari. */
window.AVJ_PREISE = window.AVJ_PREISE || (function(){
  "use strict";
  var URL_JSON  = "https://avj-tools.rent-in-nom.de/preise.json";
  var CACHE_KEY = "avjWebPreise_v1";
  var daten = null, fertig = false, warteschlange = [];

  function pruefe(d){
    if(!d || typeof d !== "object") return false;

    /* Frueher stand hier "neunsitzer und transporter muessen da sein".
       Das war zu eng: sobald ein Rechner eine andere Gruppe liest, haengt
       die Gueltigkeit der ganzen Datei an zwei fremden Abschnitten.
       Jetzt reicht: mindestens eine Gruppe mit mindestens einem Fahrzeug. */
    var gefunden = 0;
    for(var g0 in d){
      if(g0 === "version" || g0 === "stand" || g0 === "erzeugtVon") continue;
      if(d[g0] && typeof d[g0] === "object"){
        for(var k0 in d[g0]){ gefunden++; break; }
      }
    }
    if(!gefunden) return false;
    var ok = true;

    function zahlenreihe(a, laenge){
      if(!a || a.length !== laenge) return false;
      for(var i=0;i<a.length;i++){
        if(typeof a[i] !== "number" || !isFinite(a[i]) || a[i] < 0) return false;
      }
      return true;
    }
    function paarliste(a){
      if(!a || !a.length) return false;
      for(var i=0;i<a.length;i++){
        if(!a[i] || a[i].length !== 2) return false;
        if(typeof a[i][0] !== "number" || typeof a[i][1] !== "number") return false;
        if(!isFinite(a[i][0]) || !isFinite(a[i][1]) || a[i][0] < 0 || a[i][1] < 0) return false;
      }
      return true;
    }

    for(var g in d){
      if(g === "version" || g === "stand" || g === "erzeugtVon") continue;
      if(!d[g] || typeof d[g] !== "object") continue;
      for(var k in d[g]){
        var c = d[g][k];
        if(!c || typeof c !== "object"){ ok = false; continue; }
        if(!zahlenreihe(c.tier, 7))   ok = false;
        if(!zahlenreihe(c.tierKm, 7)) ok = false;
        if(c.dayMoDo && !paarliste(c.dayMoDo)) ok = false;
        if(c.dayFrSa && !paarliste(c.dayFrSa)) ok = false;
        if(c.weekend && !paarliste(c.weekend)) ok = false;
        if(c.sb){
          if(!c.sb.length) ok = false;
          for(var s=0;s<(c.sb||[]).length;s++){
            if(!c.sb[s] || !zahlenreihe(c.sb[s].tier, 7)) ok = false;
          }
        }
      }
    }
    return ok;
  }

  function melde(){
    fertig = true;
    var f = warteschlange; warteschlange = [];
    for(var i=0;i<f.length;i++){ try{ f[i](daten); }catch(e){} }
  }

  function ausCache(){
    try{
      var raw = window.localStorage.getItem(CACHE_KEY);
      if(!raw) return false;
      var d = JSON.parse(raw);
      if(!pruefe(d)) return false;
      daten = d;
      return true;
    }catch(e){ return false; }
  }

  function start(){
    /* Zeitstempel je Minute: hebelt den Zwischenspeicher von GitHub Pages
       aus, ohne ihn ganz abzuschalten. */
    var url = URL_JSON + "?t=" + Math.floor(new Date().getTime()/60000);
    var erledigt = false;
    var frist = setTimeout(function(){
      if(erledigt) return;
      erledigt = true; ausCache(); melde();
    }, 3000);

    try{
      var x = new XMLHttpRequest();
      x.open("GET", url, true);
      x.onreadystatechange = function(){
        if(x.readyState !== 4 || erledigt) return;
        erledigt = true; clearTimeout(frist);
        var d = null;
        try{ d = JSON.parse(x.responseText); }catch(e){}
        if(x.status >= 200 && x.status < 400 && pruefe(d)){
          daten = d;
          try{ window.localStorage.setItem(CACHE_KEY, JSON.stringify(d)); }catch(e){}
        } else {
          ausCache();
        }
        melde();
      };
      x.onerror = function(){
        if(erledigt) return;
        erledigt = true; clearTimeout(frist); ausCache(); melde();
      };
      x.send();
    }catch(e){
      clearTimeout(frist);
      if(!erledigt){ erledigt = true; ausCache(); melde(); }
    }
  }
  start();

  return {
    bereit: function(fn){
      if(fertig){ try{ fn(daten); }catch(e){} }
      else warteschlange.push(fn);
    },
    daten:   function(){ return daten; },
    /* Kaution: kommt aus preise.json, falls dort hinterlegt — sonst 300 */
    kaution: function(){
      var k = daten && daten.kaution;
      return (typeof k === "number" && isFinite(k) && k > 0) ? k : 300;
    }
  };
})();

/* ─── GA4-Events ─────────────────────────────────────────────
   Feuert nur, wenn Analytics geladen ist. Ohne GA laeuft alles normal. */
function avjtTrack(name, params){
  try{
    if(typeof window.gtag === "function"){
      window.gtag("event", name, params || {});
    }
  }catch(e){}
}

/* Popup-Steuerung */
(function(){
  "use strict";
  var ov    = document.getElementById("avjTOverlay");
  var open  = document.getElementById("avjTOpen");
  var close = document.getElementById("avjTClose");
  if(!ov || !open) return;
  var lastFocus = null;

  if(ov.parentNode !== document.body){ document.body.appendChild(ov); }

  function show(){
    lastFocus = document.activeElement;
    ov.hidden = false;
    avjtTrack("rechner_geoeffnet", { kategorie: "Transporter" });
    document.body.style.overflow = "hidden";
    var first = ov.querySelector(".avjt-car");
    if(first) first.focus();
  }
  function hide(){
    ov.hidden = true;
    document.body.style.overflow = "";
    if(lastFocus) lastFocus.focus();
  }
  open.addEventListener("click", show);
  if(close) close.addEventListener("click", hide);
  ov.addEventListener("click", function(e){ if(e.target.hasAttribute("data-close")) hide(); });
  document.addEventListener("keydown", function(e){ if(e.key === "Escape" && !ov.hidden) hide(); });
  ov.addEventListener("keydown", function(e){
    if(e.key !== "Tab") return;
    var f = ov.querySelectorAll('button, input, a[href], select, [tabindex]:not([tabindex="-1"])');
    if(!f.length) return;
    var first = f[0], last = f[f.length-1];
    if(e.shiftKey && document.activeElement === first){ e.preventDefault(); last.focus(); }
    else if(!e.shiftKey && document.activeElement === last){ e.preventDefault(); first.focus(); }
  });
})();

  var eur = function(n){ return Math.round(n).toLocaleString("de-DE"); };
  var num = function(n){ return Math.round(n).toLocaleString("de-DE"); };


  /* ═══ Langzeit ══════════════════════════════════════════════════
     Vier Wochen sind die Grenze. Bis dahin wird zwischen Wochen- und
     Monatsanker interpoliert, danach gibt es keinen Automatikpreis mehr.
     Wortgleich mit dem Tool — wer hier etwas aendert, aendert es dort
     auch, sonst rechnet die Website anders als der Betrieb. */
  var LZ_TAGE = 28;
  function mRund5(n){  return Math.max(5,  Math.round(n / 5)  * 5);  }
  function mRund50(n){ return Math.max(50, Math.round(n / 50) * 50); }
  function lzPreis(car){
    if(car.monthPrice > 0) return car.monthPrice;
    return mRund5((car.tier ? car.tier[6] : 0) * 3);
  }
  function lzKm(car){
    if(car.monthKm > 0) return car.monthKm;
    return mRund50((car.tierKm ? car.tierKm[6] : 0) * 2);
  }
  function lzSb(stufe){
    if(!stufe || !stufe.tier) return 0;
    if(stufe.monat !== undefined && stufe.monat !== null) return stufe.monat;
    return Math.round(stufe.tier[6] * 3);
  }
  function lzAnteil(tage){ return (tage - 7) / (LZ_TAGE - 7); }

  /* ─── Datum / Dauer ──────────────────────────────────────── */
  function parseDT(dEl, tEl){
    var d = $(dEl).value, t = $(tEl).value || "09:00";
    if(!d) return null;
    return new Date(d + "T" + t + ":00");
  }

  function duration(){
    var a = parseDT("avjTFrom","avjTFromT");
    var b = parseDT("avjTTo","avjTToT");
    if(!a || !b) return null;
    var ms = b - a;
    if(ms <= 0) return { invalid:true };
    var hours = ms / 3600000;
    return {
      from: a, to: b, hours: hours,
      days: Math.max(1, Math.ceil(hours / 24)),
      fullDays: Math.floor(hours / 24),
      restHours: Math.round(hours - Math.floor(hours/24)*24)
    };
  }

  // Wochenendtarif: Abholung Fr ab 12:00 oder Sa, Rückgabe bis Mo 10:00
  function isWeekend(d){
    if(!d || d.invalid) return false;
    var wdA = d.from.getDay();           // 5 = Fr, 6 = Sa
    var wdB = d.to.getDay();             // 0 = So, 1 = Mo
    var hA  = d.from.getHours() + d.from.getMinutes()/60;
    var hB  = d.to.getHours()   + d.to.getMinutes()/60;
    if(d.hours > 96) return false;       // l\u00E4nger als Fr\u2013Di \u2192 normale Staffel
    var startOk = (wdA === 5 && hA >= 11) || wdA === 6;
    /* Rueckgabe Montag frueh -> Wochenende.
       Rueckgabe Sonntag nur, wenn die Miete wirklich ueber das Wochenende laeuft;
       eine reine 24-Stunden-Miete Sa->So ist eine Tagesmiete. */
    var endOk   = (wdB === 1 && hB <= 11) || (wdB === 0 && d.hours > 30);
    return startOk && endOk;
  }

  /* ─── km-Stufe wählen ────────────────────────────────────── */
  /* Zwischen je zwei Paketstufen die Mitte einziehen:
     aus 100/75 und 400/115 wird zusaetzlich 250/95.
     Entspricht der bisherigen Handrechnung (beide Pakete addieren, halbieren). */
  function expand(steps){
    var out = [];
    for(var i=0;i<steps.length;i++){
      out.push(steps[i]);
      if(i+1 < steps.length){
        out.push([
          Math.round((steps[i][0] + steps[i+1][0]) / 2),
          Math.round((steps[i][1] + steps[i+1][1]) / 2)
        ]);
      }
    }
    return out;
  }

  var _gridCache = {};
  function gridFor(car, key){
    var id = car.name + "|" + key;
    if(!_gridCache[id]) _gridCache[id] = expand(car[key]);
    return _gridCache[id];
  }

  /* Guenstigste Variante fuer den Kunden waehlen: entweder das naechstgroessere
     Paket, oder ein kleineres Paket plus vorab geplante Mehrkilometer.
     Ohne km-Angabe wird die kleinste Stufe gezeigt. */
  function pickStep(steps, km, rate){
    if(km === null) return { idx:0, km:steps[0][0], price:steps[0][1] };
    if(!rate){
      for(var i=0;i<steps.length;i++){
        if(km <= steps[i][0]) return { idx:i, km:steps[i][0], price:steps[i][1] };
      }
      var last = steps[steps.length-1];
      return { idx:steps.length-1, km:last[0], price:last[1] };
    }
    var best = null;
    for(var j=0;j<steps.length;j++){
      var over = Math.max(0, km - steps[j][0]);
      var cost = steps[j][1] + over * rate;
      if(best === null || cost < best.cost - 0.001){
        best = { idx:j, km:steps[j][0], price:steps[j][1], cost:cost };
      }
    }
    return best;
  }

      badge = hatFrSa ? (frSa ? "Fr/Sa" : "Mo-Do") : "";

    } else {
      mode = "tier";
      var i = Math.min(d.days, 7) - 1;
      base   = car.tier[i];
      freeKm = car.tierKm[i];
      if(d.days >= LZ_TAGE && car.lzAnfrage){
        /* Der Haken versteckt den Vier-Wochen-Preis — nicht die Wochen
           davor. Zwei und drei Wochen rechnen weiter: dort ist der
           Monatswert nur Zwischenziel der Staffel, nie die Ausgabe. */
        return { anfrage:true, car:car, d:d, tage:d.days };
      }

        /* Ueber vier Wochen kein Automatikpreis mehr — das verhandelt
           Niran persoenlich. */
        return { anfrage:true, car:car, d:d, tage:d.days };
      }
      if(d.days > 7){
        var anteil = lzAnteil(d.days);
        base   = mRund5( car.tier[6]   + anteil * (lzPreis(car) - car.tier[6]));
        freeKm = mRund50(car.tierKm[6] + anteil * (lzKm(car)    - car.tierKm[6]));
      }
      if(hatSb){
        var stufe = car.sb[sbIdx], sbTier = stufe.tier;
        sbCost = sbTier[Math.min(d.days, 7) - 1];
        if(d.days > 7){

        }
      } else {
        sbCost = 0;
      }
      tarifLabel = d.days + (d.days === 1 ? " Tag" : " Tage");
      if(d.days === 7)                   badge = "Wochentarif";
      if(d.days > 7 && d.days < LZ_TAGE) badge = "Mehrwochentarif";
      if(d.days === LZ_TAGE)             badge = "Monatstarif";
    }

    var overKm   = km === null ? 0 : Math.max(0, km - freeKm);
    var overCost = overKm * car.planKm;
    var total    = base + sbCost + overCost;

    return {
      car: car, d: d, mode: mode, km: km,
      base: base, freeKm: freeKm, sbCost: sbCost,
      overKm: overKm, overCost: overCost, total: total,
      tarifLabel: tarifLabel, badge: badge
    };
  }

  /* Die Kennzeichnung traegt mal zwei Zeichen ("Fr/Sa"), mal zwanzig

  /* ── Feiertage Niedersachsen ────────────────────────────────
     Ostersonntag nach Gauss/Meeus, die beweglichen Tage davon abgeleitet. */
  function osterSonntag(y){
    var a=y%19, b=Math.floor(y/100), cc=y%100;
    var dd=Math.floor(b/4), e=b%4, f=Math.floor((b+8)/25);
    var g=Math.floor((b-f+1)/3), h=(19*a+b-dd-g+15)%30;
    var i=Math.floor(cc/4), k=cc%4, l=(32+2*e+2*i-h-k)%7;
    var m=Math.floor((a+11*h+22*l)/451);
    var mon=Math.floor((h+l-7*m+114)/31), day=((h+l-7*m+114)%31)+1;
    return new Date(y, mon-1, day);
  }
  var _feierCache = {};
  function feiertage(y){
    if(_feierCache[y]) return _feierCache[y];
    var p2 = function(n){ return ("0"+n).slice(-2); };
    var key = function(d){ return d.getFullYear()+"-"+p2(d.getMonth()+1)+"-"+p2(d.getDate()); };
    var o = osterSonntag(y);
    var add = function(n){ var x = new Date(o.getTime()); x.setDate(x.getDate()+n); return key(x); };
    var m = {};
    m[y+"-01-01"]            = "Neujahr";
    m[add(-2)]               = "Karfreitag";
    m[add(1)]                = "Ostermontag";
    m[y+"-05-01"]            = "Tag der Arbeit";
    m[add(39)]               = "Christi Himmelfahrt";
    m[add(50)]               = "Pfingstmontag";
    m[y+"-10-03"]            = "Tag der Deutschen Einheit";
    m[y+"-10-31"]            = "Reformationstag";
    m[y+"-12-25"]            = "1. Weihnachtstag";
    m[y+"-12-26"]            = "2. Weihnachtstag";
    _feierCache[y] = m;
    return m;
  }
  function feiertagName(ds){
    if(!ds || ds.length < 10) return null;
    var y = parseInt(ds.slice(0,4), 10);
    if(!y) return null;
    return feiertage(y)[ds] || null;
  }

  /* Heiligabend und Silvester: keine Feiertage, aber verkuerzt geoeffnet */
  function kurztagName(ds){
    if(!ds || ds.length < 10) return null;
    var md = ds.slice(5);
    if(md === "12-24") return "Heiligabend";
    if(md === "12-31") return "Silvester";
    return null;
  }

  var WOCHENTAG = ["Sonntag","Montag","Dienstag","Mittwoch","Donnerstag","Freitag","Samstag"];

  var HOURS = {
    1:[8,18], 2:[8,18], 3:[8,18], 4:[8,18], 5:[8,18],   /* Mo-Fr */
    6:[8,13],                                            /* Sa    */
    0:null                                               /* So    */
  };

  function hoursFor(dateStr){
    if(!dateStr) return HOURS[1];
    if(feiertagName(dateStr)) return null;      /* Feiertag: geschlossen */
    var wd = new Date(dateStr + "T12:00:00").getDay();
    var base = HOURS[wd];
    if(!base) return null;                      /* Sonntag bleibt zu */
    if(kurztagName(dateStr)) return [8, 13];    /* verkuerzt wie samstags */
    return base;
  }

  function minutesOf(hhmm){
    var p = hhmm.split(":");
    return parseInt(p[0],10)*60 + parseInt(p[1],10);
  }

  function pad2(n){ return ("0"+n).slice(-2); }

  /* Liste aufbauen. mode "start": nur Oeffnungszeiten.
     mode "end": voller Tag, ausserhalb der Zeiten als Tresor markiert. */
  function fillTimes(sel, dateStr, mode){
    var h = hoursFor(dateStr);
    var keep = sel.value;
    var html = "", first = null;

    if(mode === "start"){
      if(!h){ sel.innerHTML = '<option value="">geschlossen</option>'; return; }
      for(var m = h[0]*60; m <= h[1]*60; m += 30){
        var v = pad2(Math.floor(m/60)) + ":" + pad2(m%60);
        if(first === null) first = v;
        html += '<option value="'+v+'">'+v+'</option>';
      }
    } else {
      for(var t = 0; t < 24*60; t += 30){
        var v2 = pad2(Math.floor(t/60)) + ":" + pad2(t%60);
        var inside = h && t >= h[0]*60 && t <= h[1]*60;
        if(first === null) first = v2;
        html += '<option value="'+v2+'"'+(inside?"":' class="tresor"')+'>'
             +  v2 + (inside ? "" : " \u00B7 Tresor") + '</option>';
      }
    }
    sel.innerHTML = html;
    /* bisherige Auswahl halten, sonst erste gueltige Zeit */
    var ok = false;
    for(var i=0;i<sel.options.length;i++) if(sel.options[i].value === keep) ok = true;
    sel.value = ok ? keep : (mode === "start" ? (first || "") : (keep || "10:00"));
    if(!sel.value && sel.options.length) sel.value = sel.options[0].value;
  }


  function setWd(id, dateStr){
    var el = $(id);
    if(!el) return;
    if(!dateStr || dateStr.length < 10){ el.textContent = ""; el.className = "wd"; return; }
    var d = new Date(dateStr + "T12:00:00");
    var f = feiertagName(dateStr);
    el.textContent = WOCHENTAG[d.getDay()];
    el.className = "wd" + ((f || d.getDay() === 0) ? " zu" : "");
    el.title = f || "";
  }

  function refreshTimes(){
    var fromEl = $("avjTFromT"), toEl = $("avjTToT");
    fillTimes(fromEl, $("avjTFrom").value, "start");
    fillTimes(toEl,   $("avjTTo").value,   "end");
    setWd("avjTFromWd", $("avjTFrom").value);
    setWd("avjTToWd",   $("avjTTo").value);

    /* Hinweis Abholung */
    var hF = hoursFor($("avjTFrom").value);
    var eF = $("avjTFromHint");
    if(!hF){
      var fnF = feiertagName($("avjTFrom").value);
      eF.textContent = fnF
        ? (fnF + " \u2013 an diesem Tag keine Abholung m\u00F6glich.")
        : "Sonntags keine Abholung \u2013 bitte Samstag abholen.";
      eF.className = "h warn";
    } else {
      var ktF = kurztagName($("avjTFrom").value);
      eF.textContent = (ktF ? ktF + " \u00B7 " : "") +
                       "Abholung " + pad2(hF[0]) + ":00\u2013" + pad2(hF[1]) + ":00 Uhr";
      eF.className = "h";
    }

    /* Hinweis Rueckgabe */
    var hT = hoursFor($("avjTTo").value);
    var eT = $("avjTToHint");
    var mT = minutesOf(toEl.value || "10:00");
    var insideT = hT && mT >= hT[0]*60 && mT <= hT[1]*60;
    if(insideT){
      eT.textContent = "R\u00FCckgabe pers\u00F6nlich m\u00F6glich";
      eT.className = "h";
    } else {
      eT.textContent = "Ausserhalb der \u00D6ffnungszeiten \u2013 R\u00FCckgabe \u00FCber Schl\u00FCsseltresor";
      eT.className = "h tresor";
    }
  }

  /* Startwert: naechster Tag mit Oeffnungszeiten */
  (function(){
    var d = new Date();
    var guard = 0;
    while(!HOURS[d.getDay()] && guard < 7){ d.setDate(d.getDate()+1); guard++; }
    $("avjTFrom").value = d.getFullYear()+"-"+pad2(d.getMonth()+1)+"-"+pad2(d.getDate());
  })();
  refreshTimes();

  render();
